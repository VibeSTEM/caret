cmake_minimum_required(VERSION 3.22)
project(caret_gtest_consumer_fixture LANGUAGES NONE)
if(CARET_FIXTURE_LINK_TEST)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  if(WIN32)
    add_compile_definitions(_ITERATOR_DEBUG_LEVEL=0)
  endif()
  include(CTest)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../../sdk-provision.cmake")
if(CARET_FIXTURE_REQUEST_GTEST)
  caret_require_googletest()
  if(BUILD_TESTING AND (NOT TARGET GTest::gtest OR NOT TARGET GTest::gtest_main))
    message(FATAL_ERROR "真实测试消费者未得到 GoogleTest targets")
  endif()
  # 同一 configure 内重复请求必须复用同一 SDK，不能产生第二份 target。
  caret_require_googletest()
endif()

if(CARET_FIXTURE_LINK_TEST AND BUILD_TESTING)
  add_executable(caret_gtest_consumer
                 "${CMAKE_CURRENT_LIST_DIR}/../../../../../tools/sdk/selfcheck/googletest/main.cpp")
  target_link_libraries(caret_gtest_consumer PRIVATE GTest::gtest_main)
  add_test(NAME GoogleTest.ConsumerSmoke COMMAND caret_gtest_consumer)
endif()
